def sha(b):
    return hashlib.sha256(b).hexdigest()


def gitblob(b):
    return hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest()


def finite(x):
    return isinstance(x, (int, float, np.integer, np.floating)) and not isinstance(x, (bool, np.bool_)) and math.isfinite(x)


def tau(x, y):
    r = kendalltau(x, y).statistic
    return float(r) if math.isfinite(r) else None


def numeric_mean(xs):
    v = [float(x) for x in xs if finite(x)]
    return float(np.mean(v)) if v else None


def quantiles(xs):
    x = [v for v in xs if finite(v)]
    return [float(v) for v in np.quantile(x, [0, .25, .5, .75, 1])] if x else None

def rank_interval(value, population, high=False):
    v = np.array(population)
    return {'best': 1+int(np.sum(v > value if high else v < value)),
            'worst': int(np.sum(v >= value if high else v <= value)), 'of':len(v)}


def rank_changes(original, alternative):
    x = np.asarray(original); y = np.asarray(alternative)
    pairs = comparable = flipped = new_ties = 0
    for i in range(len(x)):
        for j in range(i+1,len(x)):
            pairs += 1
            a = np.sign(x[i]-x[j]); b = np.sign(y[i]-y[j])
            comparable += bool(a)
            flipped += bool(a*b < 0)
            new_ties += bool(a and not b)
    return {'tau_b':tau(x,y),'all_pairs':pairs,'original_strict_pairs':comparable,
            'strict_reversals':flipped,'strict_reversal_fraction':flipped/comparable if comparable else None,
            'strict_order_to_tie':new_ties}


def historical_parts(o):
    kind = o['kind']
    parts = {'failure':0., 'feasible_regret':0., 'missed_opportunity':0., 'correct_abstention':0.}
    if kind == 'SELECTED_INFEASIBLE': parts['failure']=10.
    elif kind == 'SELECTED_FEASIBLE': parts['feasible_regret']=o['gap_to_best_in_set_s']/UNIT
    elif kind == 'MISSED_OPPORTUNITY': parts['missed_opportunity']=1.
    elif kind == 'CORRECT_ABSTENTION': pass
    else:
        assert o['decision_loss'] is None
        return None
    assert abs(sum(parts.values())-o['decision_loss']) < 1e-9
    return parts


def component_analysis(data, members, questions):
    loss=[]; matrix=[]; gain=[]; identity_n=0; kinds=collections.Counter()
    for m in members:
        rows=[data['decisions'][m][q]['outcome'] for q in questions]
        parts=[historical_parts(o) for o in rows]
        assert all(p is not None for p in parts)
        loss.append(float(np.mean([o['decision_loss'] for o in rows])))
        matrix.append([float(np.mean([p[k] for p in parts])) for k in ['failure','feasible_regret','missed_opportunity']])
        gains=[]
        for o in rows:
            kinds[o['kind']] += 1
            if o['kind']=='SELECTED_FEASIBLE' and o['improvement_vs_baseline_s'] is not None:
                base=o['verified_objective_s']+o['improvement_vs_baseline_s']
                assert abs((base-o['best_objective_s'])-o['improvement_vs_baseline_s']-o['gap_to_best_in_set_s'])<1e-8
                gains.append(o['improvement_vs_baseline_s']/UNIT); identity_n += 1
        gain.append(numeric_mean(gains))
    x=np.asarray(matrix); l=np.asarray(loss); total=float(np.var(l))
    shares={k:float(np.mean(x[:,i]))/float(np.mean(l)) if np.mean(l)>0 else None for i,k in enumerate(['failure','feasible_regret','missed_opportunity'])}
    variance={k:float(np.cov(x[:,i],l,ddof=0)[0,1])/total if total else None for i,k in enumerate(['failure','feasible_regret','missed_opportunity'])}
    assert np.max(np.abs(x.sum(axis=1)-l))<1e-9
    if total: assert abs(sum(variance.values())-1)<1e-8
    return {'members':len(members),'questions':questions,'question_count':len(questions),'outcome_counts':dict(kinds),
            'mean_loss':float(np.mean(l)),'loss_magnitude_shares':shares,
            'variance_accounting_cov_component_with_total_over_var_total':variance,
            'tau_loss_with_unsafe_pick_rate':tau(l,x[:,0]),
            'component_only_tau':{k:tau(l,x[:,i]) for i,k in enumerate(['failure','feasible_regret','missed_opportunity'])},
            'leave_component_out':{k:rank_changes(l,l-x[:,i]) for i,k in enumerate(['failure','feasible_regret','missed_opportunity'])},
            'verified_gain_vs_registered_default_mean_by_member_quantiles':quantiles(gain),
            'verified_baseline_gain_identity_checked_rows':identity_n,
            'per_member':{m:{'loss':loss[i],'failure':matrix[i][0],'feasible_regret':matrix[i][1],'missed_opportunity':matrix[i][2],
                             'mean_gain_vs_registered_default_when_available':gain[i]} for i,m in enumerate(members)}}

def order_and_outcome(q,pred,truth):

def utility(o,qi,lam,kappa=0):
    if not finite(o['decision_loss']): return None
    unsafe=o['kind']=='SELECTED_INFEASIBLE'
    attempted=o['selected'] is not None
    if qi['fallback'] is None:
        if not qi['settled_none_feasible']: return None
        assert o['kind'] in ('CORRECT_ABSTENTION','SELECTED_INFEASIBLE')
        return -lam*unsafe-kappa*attempted
    gain=(qi['fallback_s']-o['verified_objective_s'])/UNIT if o['kind']=='SELECTED_FEASIBLE' else 0.
    return gain-lam*unsafe-kappa*attempted


def verify(order,q,truth,k):
    attempted=order[:k]
    assert len(attempted)<=k and len(set(attempted))==len(attempted)
    feasible=[a for a in attempted if truth[a]['status']==decision.FEASIBLE]
    chosen=min(feasible,key=lambda a:(truth[a]['objective'],BYID[a]['c1'],BYID[a]['c2'],a)) if feasible else None
    qi=QINFO[q['id']]
    gain=max(0.,(qi['fallback_s']-truth[chosen]['objective'])/UNIT) if chosen and qi['fallback'] else 0.
    best_s=qi['best_s']
    service_loss=min(1.,(truth[chosen]['objective']-best_s)/UNIT) if chosen and best_s is not None else 1.
    rejected=sum(truth[a]['status']==decision.INFEASIBLE for a in attempted)
    unknown=sum(truth[a]['status'] in (decision.UNRESOLVED,decision.UNAVAILABLE) for a in attempted)
    proxy_seconds=0.; proxy_known=True
    for a in attempted:
        for t,s in q['conditions']:
            r=REFS[REF_INDEX[BYID[a]['c1'],BYID[a]['c2'],t,s]]
            if not finite(r.get('wall_s')): proxy_known=False
            else: proxy_seconds+=r['wall_s']
    if chosen: assert truth[chosen]['status']==decision.FEASIBLE
    return {'returned':chosen,'new_gain_units':gain,'service_loss':service_loss,'attempts':len(attempted),
            'complete_case_tokens':len(attempted)*len(q['conditions']),'unsafe_rejections':rejected,'unresolved_or_unavailable_attempts':unknown,
            'proxy_reference_wall_s':proxy_seconds if proxy_known else None}


def controlled_order(q,kind,guard):
    truth=TRUTH[q['id']]
    if kind=='always_abstain': return []
    if kind=='always_default': return [BASE]
    if kind=='guarded_default':
        return [BASE] if all(t>=guard['t_min'] and s>=guard['soc_min'] for t,s in q['conditions']) else []
    if kind=='oracle_filtered_default': return [BASE] if truth[BASE]['status']==decision.FEASIBLE else []
    if kind=='oracle_slowest_feasible':
        fs=[c for c in CAND if truth[c['id']]['status']==decision.FEASIBLE]
        return [max(fs,key=lambda c:(truth[c['id']]['objective'],-c['c1'],-c['c2']))['id']] if fs else []
    raise ValueError(kind)

def partial_rank(x,y,nominal,families=None):
    z=np.column_stack([np.ones(len(x)),rankdata(nominal)])
    if families is not None:
        z=np.column_stack([z]+[(np.asarray(families)==f).astype(float) for f in sorted(set(families))[:-1]])
    a=rankdata(x);b=rankdata(y)
    ar=a-z@np.linalg.lstsq(z,a,rcond=None)[0]; br=b-z@np.linalg.lstsq(z,b,rcond=None)[0]
    if np.std(ar)<1e-10 or np.std(br)<1e-10:return None
    return float(np.corrcoef(ar,br)[0,1])

def cached_scalar_order(q, method):
    """Public D reference-only scalar map, never reads V responses to order."""
    ts=[5.,14.,24.,34.]; ss=[.12,.33,.48]
    quantities={}
    for c in CAND:
        for i,(t,s) in enumerate(q['conditions']):
            if method=='nearest_D':
                td,sd=min(((a,b) for a in ts for b in ss),
                          key=lambda ab:(((t-ab[0])/35)**2+((s-ab[1])/.45)**2,ab))
                nodes=[(td,sd,1.)]
            elif method=='bilinear_D_clamped':
                tc=min(max(t,ts[0]),ts[-1]); sc=min(max(s,ss[0]),ss[-1])
                ta,tb=next((a,b) for a,b in zip(ts,ts[1:]) if a<=tc<=b)
                sa,sb=next((a,b) for a,b in zip(ss,ss[1:]) if a<=sc<=b)
                wt=(tc-ta)/(tb-ta); ws=(sc-sa)/(sb-sa)
                nodes=[(ta,sa,(1-wt)*(1-ws)),(tb,sa,wt*(1-ws)),(ta,sb,(1-wt)*ws),(tb,sb,wt*ws)]
                nodes=[n for n in nodes if n[2]>1e-15]
            else: raise ValueError(method)
            records=[(REFS[REF_INDEX[c['c1'],c['c2'],a,b]],w) for a,b,w in nodes]
            measured=[(decision.measure(CON,r['outputs']),w) for r,w in records if r['status']=='OK']
            if len(measured)!=len(records) or any(v['time_to_cv_onset_s'] is None for v,w in measured):
                # Unavailable map prediction is omitted, never extrapolated or fabricated.
                quantities[c['id'],i]=None
            else:
                quantities[c['id'],i]={key:sum(v[key]*w for v,w in measured) for key in measured[0][0]}
    admissible=[]
    for c in CAND:
        per=[quantities[c['id'],i] for i in range(len(q['conditions']))]
        if any(v is None for v in per): continue
        checks=[decision.check(CON,v) for v in per]
        if all(v==decision.PASS for ck in checks for v in ck.values()):
            admissible.append((max(v['time_to_cv_onset_s'] for v in per),c['c1'],c['c2'],c['id']))
    return [v[-1] for v in sorted(admissible)]
