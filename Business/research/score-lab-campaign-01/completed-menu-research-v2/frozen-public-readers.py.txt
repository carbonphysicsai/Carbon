def sha(data): return hashlib.sha256(data).hexdigest()
def pinned(path,digest):

def skip(text,i):
    """Skip JSON without decoding non-allowlisted outcome values."""
    while text[i].isspace(): i+=1
    if text[i] not in '[{"':
        while i<len(text) and text[i] not in ',}]': i+=1
        return i
    quoted=text[i]=='"'; escaped=False; stack=[]
    if not quoted: stack.append(text[i])
    i+=1
    while i<len(text):
        c=text[i]
        if quoted:
            if escaped: escaped=False
            elif c=='\\': escaped=True
            elif c=='"':
                quoted=False
                if not stack: return i+1
        elif c=='"': quoted=True
        elif c in '[{': stack.append(c)
        elif c in ']}':
            if not stack or (stack.pop(),c) not in (('[',']'),('{','}')):
                raise ValueError('JSON_DELIMITER')
            if not stack: return i+1
        i+=1
    raise ValueError('JSON_TRUNCATED')

def fields(text,start=0):
    dec=json.JSONDecoder();i=start
    while text[i].isspace():i+=1
    if text[i]!='{':raise ValueError('JSON_OBJECT_REQUIRED')
    i+=1;seen=set()
    while True:
        while text[i].isspace() or text[i]==',':i+=1
        if text[i]=='}':return
        key,i=dec.raw_decode(text,i)
        if not isinstance(key,str) or key in seen:raise ValueError('JSON_DUPLICATE_KEY')
        seen.add(key)
        while text[i].isspace():i+=1
        if text[i]!=':':raise ValueError('JSON_COLON')
        i+=1
        while text[i].isspace():i+=1
        end=skip(text,i)
        yield key,i,end
        i=end

def predictions(path,digest,wanted,member):

    if len(data)>MAX_EXPANDED:raise ValueError('GZIP_EXPANSION_CAP')
    text=data.decode('utf-8');dec=json.JSONDecoder();out={};found=None
    for key,start,_ in fields(text):
        if key=='member':found=dec.raw_decode(text,start)[0]
        elif key=='predictions':
            for cid,a,_ in fields(text,start):
                if cid in wanted:out[cid]=dec.raw_decode(text,a)[0]
    if found!=member:raise ValueError('MODEL_IDENTITY')
    return out

def reference_rows(path,wanted):
    raw=pinned(path,OLD_REF_SHA);out={}
    for line in raw.decode('utf-8').splitlines():
        m=CASE_ID.search(line)
        if not m or m[1] not in wanted:continue
        row=json.loads(line)
        if row.get('refined'):continue
        if row['case_id'] in out:raise ValueError('D_REFERENCE_DUPLICATE')
        out[row['case_id']]=row
    if set(out)!=wanted:raise ValueError('D_REFERENCE_COVERAGE')
    return out

def value_losses(utilities,question_ids):
    """One unknown verified outcome withholds the whole fixed VALUE vector."""
    losses={};unknown={};accounting={}
    for member,rows in utilities.items():
        if rows is None:
            unknown[member]={'reason':'VALUE_PREDICTION_COVERAGE_UNKNOWN','questions':list(question_ids)}
            continue
        unresolved=[q for q in question_ids if rows[q]['unresolved_attempts']>0]
        if unresolved:
            unknown[member]={'reason':'VALUE_VERIFICATION_REFERENCE_UNRESOLVED','questions':unresolved}
            accounting[member]={q:{'n':rows[q]['n'],'unresolved_attempts':rows[q]['unresolved_attempts'],'U':None,'G':None} for q in question_ids}
            continue
        losses[member]=[-rows[q]['U'] for q in question_ids]
    return losses,unknown,accounting
