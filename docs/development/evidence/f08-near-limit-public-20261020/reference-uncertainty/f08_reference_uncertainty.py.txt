"""F08 reference uncertainty from the RR-PACKS-07 ladder (Validator, 2026-10-20:
"apply the Motor method as written"; MOTOR_REF_UNCERTAINTY_METHOD_20261019).

    python3 f08_reference_uncertainty.py LADDER_OBSERVED.json > record.json

One adaptation, forced by the data and stated in the record: the Motor ladder
is ratio 2, F08's rungs are h = 3, 2.25, 1.7 mm (ratios 1.333 and 1.324). The
rule is Roache's non-uniform form (Celik et al. 2008), which reduces exactly
to the written one at r = 2. With phi1 (h1.7), phi2 (h2.25), phi3 (h3),
e21 = phi2 - phi1, e32 = phi3 - phi2, r21 = h2/h1, r32 = h3/h2, s = sign(e32/e21):
  p  = |ln|e32/e21| + q(p)| / ln r21,  q(p) = ln((r21^p - s) / (r32^p - s));
  fine-level estimate = |e21| / (r21^p - 1).
U_q = the largest fine-level estimate over the panel computed with the LOWEST
observed p on the panel, rounded up to 2 significant digits; positive by
construction. Angular term: not applicable (no angular sampling in F08).
Stop triggers, as written (applied to every panel case, all near-limit by design): p < 0.5 or non-monotone
differences; a near-limit label at the finest level differing from the
standard-mesh (h3) label beyond U_q; angular dominance n/a.
"""

import hashlib
import json
import math
import sys

QUANTITIES = {
    "worst_dynamic_mm_per_n": {"limit": 0.15, "feasible": "<="},
    "static_tip_mm_per_n": {"limit": 0.03, "feasible": "<="},
}
RUNGS = ("", "-h2.25", "-h1.7")  # h3 (standard), h2.25, h1.7 (finest)


def order(phi3, phi2, phi1, h3, h2, h1):
    e32, e21 = phi3 - phi2, phi2 - phi1
    if e21 == 0 or e32 == 0:
        return None, e32, e21
    r21, r32 = h2 / h1, h3 / h2
    s = 1.0 if e32 / e21 > 0 else -1.0
    p = abs(math.log(abs(e32 / e21))) / math.log(r21)
    for _ in range(200):
        q = math.log((r21**p - s) / (r32**p - s))
        nxt = abs(math.log(abs(e32 / e21)) + q) / math.log(r21)
        if abs(nxt - p) < 1e-12:
            p = nxt
            break
        p = nxt
    return p, e32, e21


def round_up_2sig(x):
    if x <= 0:
        return x
    e = math.floor(math.log10(x)) - 1
    return math.ceil(x / 10**e) * 10**e


def label(value, rule):
    return "feasible" if value <= rule["limit"] else "infeasible"


def record(observed):
    panel, per_case = sorted(observed["convergence"]), {}
    for g in panel:
        rows = [observed["cases"][g + suffix] for suffix in RUNGS]
        h3, h2, h1 = (r["h_mm"] for r in rows)
        per_case[g] = {"h_mm": [h3, h2, h1]}
        for q in QUANTITIES:
            phi3, phi2, phi1 = (r[q] for r in rows)
            p, e32, e21 = order(phi3, phi2, phi1, h3, h2, h1)
            per_case[g][q] = {
                "values": [phi3, phi2, phi1],
                "p": p,
                "monotone": e32 * e21 > 0 and abs(e21) < abs(e32),
                "r21": h2 / h1,
            }
    out, triggers = {}, []
    for q, rule in QUANTITIES.items():
        orders = [per_case[g][q]["p"] for g in panel]
        if any(p is None for p in orders):
            triggers.append(f"zero_difference:{q}")
            continue
        p_low = min(orders)
        estimates = [
            abs(per_case[g][q]["values"][1] - per_case[g][q]["values"][2])
            / (per_case[g][q]["r21"] ** p_low - 1)
            for g in panel
        ]
        u = round_up_2sig(max(estimates))
        out[q] = {"U": u, "p_lowest": p_low, "largest_fine_estimate": max(estimates)}
        if p_low < 0.5:
            triggers.append(f"order_below_0.5:{q}")
        if not all(per_case[g][q]["monotone"] for g in panel):
            triggers.append(f"non_monotone:{q}")
        for g in panel:
            fine, standard = per_case[g][q]["values"][2], per_case[g][q]["values"][0]
            if label(fine, rule) != label(standard, rule) and abs(fine - rule["limit"]) > u:
                triggers.append(f"label_flip:{g}:{q}")
    body = {
        "schema": "carbon.f08.reference-uncertainty.v1",
        "scope": "DEVELOPMENT reference uncertainty; not a qualification",
        "method": "MOTOR_REF_UNCERTAINTY_METHOD_20261019, Roache non-uniform-ratio form",
        "adaptation": "F08 rungs are h = 3, 2.25, 1.7 mm (ratios 1.333, 1.324), not ratio 2; "
        "the non-uniform form reduces to the written rule at r = 2",
        "source": "RR-PACKS-07 ladder (docs/development/evidence/"
        "f08-near-limit-public-20261020/ladder-observed.json)",
        "panel": panel,
        "per_case": per_case,
        "U": {q: v["U"] for q, v in out.items()},
        "detail": out,
        "angular_term": "not applicable",
        "stop_triggers_fired": triggers,
    }
    body["sha256"] = hashlib.sha256(
        json.dumps(body, sort_keys=True, separators=(",", ":")).encode()
    ).hexdigest()
    return body


if __name__ == "__main__":
    print(json.dumps(record(json.load(open(sys.argv[1]))), indent=1, sort_keys=True))
